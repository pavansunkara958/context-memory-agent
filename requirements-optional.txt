# Optional upgrades. Install what your platform supports; each one is
# independently guarded by a fallback, so a failure here changes quality, not
# capability.
#
#   pip install -r requirements-optional.txt
#
# Or install them one at a time — a failure on the first should not stop the
# second:
#
#   pip install sentence-transformers   # semantic embeddings + cross-encoder
#   pip install chromadb                # persistent vector DB
#   pip install openai httpx            # model-written answers
#
# Check what actually took effect:
#
#   python -m src.main index            # prints store, embedder, reranker

# --- Semantic embeddings and cross-encoder reranking ------------------------
# Falls back to: deterministic hash embedder + lexical-overlap reranker.
# Impact: dense retrieval and paraphrase matching get much better.
# Needs torch. No macOS x86_64 wheels after torch 2.2 — on an Intel Mac use a
# Python 3.11 environment (see README) or accept the hash backend.
sentence-transformers>=3.0

# --- Persistent vector database ---------------------------------------------
# Falls back to: NumpyStore, exact cosine search over .npy + .json.
# Impact: none on results at this corpus size. The numpy store is exact, so it
# is if anything more accurate than HNSW — it would only lose somewhere north
# of ~100k vectors, where approximate search starts to matter.
# Needs onnxruntime, which has no cp314 wheel yet.
chromadb>=1.0.0

# --- Model-written answers ---------------------------------------------------
# Falls back to: extractive answerer (selects sentences from retrieved chunks).
# Impact: prose quality only. Retrieval, memory and every metric are unaffected.
openai>=1.60
httpx>=0.27
